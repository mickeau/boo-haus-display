# Boo Haus Display - Systemd Service File
# Save as: /etc/systemd/system/boo-haus-display.service
#
# Installation:
#   Prefer: sudo ./install_service.sh
#   Or copy this file after editing User/WorkingDirectory/ExecStart:
#   sudo cp boo-haus-display.service /etc/systemd/system/
#   sudo systemctl daemon-reload
#   sudo systemctl enable boo-haus-display
#   sudo systemctl start boo-haus-display
#
# Management:
#   sudo systemctl status boo-haus-display
#   sudo systemctl stop boo-haus-display
#   sudo systemctl restart boo-haus-display
#   sudo journalctl -u boo-haus-display -f

[Unit]
Description=Boo Haus Admin Display System
After=network-online.target
Wants=network-online.target

[Service]
Type=simple
User=YOUR_USER
Group=YOUR_USER
WorkingDirectory=/home/YOUR_USER/boo_haus_display

# Main command - continuous cycling
ExecStart=/usr/bin/python3 /home/YOUR_USER/boo_haus_display/boo_haus_main.py --cycle

# Restart policy
Restart=on-failure
RestartSec=10

# Environment
Environment=PYTHONUNBUFFERED=1
EnvironmentFile=-/home/YOUR_USER/boo_haus_display/.env

# Logging
StandardOutput=journal
StandardError=journal

# Security
NoNewPrivileges=true
PrivateTmp=true
ProtectSystem=strict
ProtectHome=read-only
ReadWritePaths=/home/YOUR_USER/boo_haus_display

# Resource limits (optional)
MemoryMax=512M
CPUQuota=50%

[Install]
WantedBy=multi-user.target
